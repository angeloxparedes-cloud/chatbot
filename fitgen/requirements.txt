# --- Embeddings & vector store ---
# torch is a dependency of sentence-transformers. By default, pip installs
# the standard PyPI build of torch, which on Linux bundles CUDA/GPU
# libraries you'll never use on a CPU-only host -- this adds several
# hundred MB to both install size and runtime memory for nothing. Pointing
# at PyTorch's own CPU-only wheel index instead installs a much smaller,
# CPU-only build with the same API, which meaningfully lowers peak memory
# usage on small hosts (e.g. free-tier services with a 512MB RAM limit).
--extra-index-url https://download.pytorch.org/whl/cpu
torch==2.5.1+cpu

sentence-transformers==3.3.1
faiss-cpu==1.15.0
